async def handle_registration_event(payload: RegistrationPayload):
    # 1. Upsert contact record
    user = await db.users.upsert(
        where={ "phone": payload.phone },
        update={ "name": payload.name, "email": payload.email },
        create={ "phone": payload.phone, "name": payload.name, "whatsapp_status": 'UNKNOWN' }
    )

    # 2. Check Idempotency before message dispatch
    idempotency_key = f"REG_{payload.webinar_id}_{user.user_id}"
    if await redis.exists(idempotency_key):
        return { "status": "SKIPPED_DUPLICATE", "key": idempotency_key }

    # 3. Community Membership Check & Template Selection
    if user.whatsapp_status != 'JOINED':
        template = "WA_COMMUNITY_ONBOARDING_V2"
        params = [user.name, payload.community_invite_url]
    else:
        template = "WA_CALENDAR_CONFIRM_DIRECT"
        params = [user.name, payload.calendar_link]
        await db.webinar_registrations.mark_calendar_sent(user.user_id)

    # 4. Dispatch via Meta Cloud API Gateway
    response = await whatsapp_gateway.send_template(user.phone, template, params)
    await db.message_logs.create(idempotency_key, response.wamid, delivery_status='SENT')
    await redis.setex(idempotency_key, 86400, "PROCESSED")

async def execute_countdown_scheduler(webinar_id: UUID, offset_min: int):
    registrants = await db.query(f"""
        SELECT r.user_id, u.phone, u.name 
        FROM webinar_registrations r
        JOIN users u ON u.user_id = r.user_id
        WHERE r.webinar_id = %s AND r.status = 'ACTIVE'
    """, [webinar_id])

    for attendee in registrants:
        idem_key = f"CD_{webinar_id}_{offset_min}_{attendee.user_id}"
        if not await redis.set(idem_key, "LOCKED", nx=True, ex=3600):
            continue
        await message_queue.enqueue(
            template=f"COUNTDOWN_{abs(offset_min)}M",
            recipient=attendee.phone,
            payload={ "user": attendee.name, "url": generate_magic_link(attendee.user_id) }
        )

async def handle_conversion_webhook(event: PaymentEvent):
    # 1. Flag User as Converted in DB
    await db.users.update(
        where={ "email": event.customer_email },
        data={ "has_purchased": True, "purchased_at": now() }
    )
    # 2. Invalidate pending queued messages in Redis
    keys = await redis.keys(f"MSG_QUEUE_{event.user_id}_*")
    if keys:
        await redis.delete(*keys)
    
    # 3. Fire Instant Onboarding WhatsApp Sequence
    await whatsapp_gateway.send_template(
        event.phone, 
        template="PURCHASE_RECEIPT_PORTAL_ACCESS",
        params=[event.amount, event.login_magic_url]
    )
